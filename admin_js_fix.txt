const labels={leadership:"Leadership",service:"Service",drill:"Drill & Color Guard",adventure:"Adventure",academics:"Academics",events:"Events"};
let items=Array.isArray(window.CALLAWAY_GALLERY)?JSON.parse(JSON.stringify(window.CALLAWAY_GALLERY)):[];
let selectedPhoto="";

const $=id=>document.getElementById(id);

function esc(s=""){
  return String(s).replace(/[&<>"']/g,m=>({"&":"&amp;","<":"&lt;",">":"&gt;","\"":"&quot;","'":"&#039;"}[m]));
}

function toast(msg,type="ok"){
  const t=$("toast");
  t.textContent=msg;
  t.style.borderColor=type==="error"?"#ff4d5e":"rgba(255,255,255,.12)";
  t.style.color=type==="error"?"#ffd4d8":"#fff";
  t.classList.add("show");
  setTimeout(()=>t.classList.remove("show"),2600);
}

function ghStatus(msg,type="ok"){
  const box=$("githubStatus");
  if(!box)return;
  box.hidden=false;
  box.className="github-status "+type;
  box.textContent=msg;
}

function formatBytes(n){
  if(n<1024)return n+" B";
  if(n<1048576)return(n/1024).toFixed(1)+" KB";
  return(n/1048576).toFixed(1)+" MB";
}

function readDataURL(file){
  return new Promise((resolve,reject)=>{
    const r=new FileReader();
    r.onerror=()=>reject(new Error("Could not read this photo."));
    r.onload=()=>resolve(r.result);
    r.readAsDataURL(file);
  });
}

async function optimizeImage(file){
  const raw=await readDataURL(file);
  try{
    const img=new Image();
    await new Promise((resolve,reject)=>{
      img.onload=resolve;
      img.onerror=reject;
      img.src=raw;
    });

    const maxW=1800,maxH=1400;
    const scale=Math.min(1,maxW/img.naturalWidth,maxH/img.naturalHeight);
    const w=Math.max(1,Math.round(img.naturalWidth*scale));
    const h=Math.max(1,Math.round(img.naturalHeight*scale));

    const c=document.createElement("canvas");
    c.width=w;c.height=h;
    const ctx=c.getContext("2d",{alpha:false});
    ctx.drawImage(img,0,0,w,h);

    return c.toDataURL("image/jpeg",.84)||raw;
  }catch(e){
    return raw;
  }
}

$("fileInput").addEventListener("change",async e=>{
  const file=e.target.files&&e.target.files[0];
  selectedPhoto="";

  if(!file){
    $("fileName").textContent="No photo selected";
    return;
  }

  $("fileName").textContent=`Preparing ${file.name} • ${formatBytes(file.size)}`;

  try{
    selectedPhoto=await optimizeImage(file);
    $("photoPreview").src=selectedPhoto;
    $("photoPreview").hidden=false;
    $("fileName").textContent=`${file.name} • READY`;
    toast("Photo ready to add");
  }catch(err){
    console.error(err);
    $("fileName").textContent="Could not read photo";
    $("photoPreview").hidden=true;
    toast("Photo could not be read","error");
  }
});

$("addBtn").addEventListener("click",()=>{
  const url=$("urlInput").value.trim();
  const image=selectedPhoto||url;
  const title=$("titleInput").value.trim();

  if(!image){
    toast("Choose a photo first","error");
    return;
  }

  });

  render();

  selectedPhoto="";
  $("fileInput").value="";
  $("fileName").textContent="No photo selected";
  $("photoPreview").src="";
  $("photoPreview").hidden=true;

  toast("Photo added to gallery");
});

function render(){
  const c=$("cards");
  c.innerHTML="";
  $("count").textContent=items.length;

  if(!items.length){
    c.innerHTML='<div style="grid-column:1/-1;padding:50px;text-align:center;color:#7890a4;border:1px dashed rgba(255,255,255,.12);border-radius:12px">No photos yet.</div>';
    return;
  }

  items.forEach((p,i)=>{
    const d=document.createElement("div");
    d.className="card";
    d.innerHTML=`
      <img src="${p.image}" alt="">
      <div class="body">
        <span>${labels[p.category]||p.category} • ${p.layout||"normal"}</span>
        <h3>${esc(p.title)}</h3>
        <p>${esc(p.caption||"")}</p>
        <div class="buttons">
          <button data-a="up">↑</button>
          <button data-a="down">↓</button>
          <button data-a="edit">EDIT</button>
          <button data-a="delete">DELETE</button>
        </div>
      </div>`;

    d.querySelector('[data-a="up"]').onclick=()=>{
      if(i>0){
        [items[i-1],items[i]]=[items[i],items[i-1]];
        render();
      }
    };

    d.querySelector('[data-a="down"]').onclick=()=>{
      if(i<items.length-1){
        [items[i+1],items[i]]=[items[i],items[i+1]];
        render();
      }
    };

    d.querySelector('[data-a="edit"]').onclick=()=>{
      const t=prompt("Title:",p.title);
      if(t===null)return;
      const cap=prompt("Caption:",p.caption||"");
      if(cap===null)return;
      p.title=t.trim()||p.title;
      p.caption=cap.trim();
      render();
    };

    d.querySelector('[data-a="delete"]').onclick=()=>{
      if(confirm(`Delete "${p.title}"?`)){
        items.splice(i,1);
        render();
      }
    };

    c.appendChild(d);
  });
}

function headers(token){
  return{
    "Accept":"application/vnd.github+json",
    "Authorization":"Bearer "+token,
    "X-GitHub-Api-Version":"2022-11-28",
    "Content-Type":"application/json"
  };
}

async function api(url,opts={}){
  let r;
  try{
    r=await fetch(url,opts);
  }catch(e){
    throw new Error("Browser could not reach GitHub API: "+e.message);
  }

  let b=null;
  try{b=await r.json()}catch(_){}

  if(!r.ok){
    let m=b&&b.message?b.message:`GitHub HTTP ${r.status}`;
    if(r.status===401)m+=" — token invalid or expired.";
    if(r.status===403)m+=" — token needs Contents: Read and write.";
    if(r.status===404)m+=" — token may not have access to Photo-Gallery.";
    throw new Error(m);
  }

  return b;
}

function b64Utf8(text){
  const bytes=new TextEncoder().encode(text);
  let bin="";
  for(let i=0;i<bytes.length;i+=32768){
    bin+=String.fromCharCode(...bytes.subarray(i,i+32768));
  }
  return btoa(bin);
}

function dataParts(data){
  const m=String(data).match(/^data:image\/([^;]+);base64,(.+)$/);
  if(!m)return null;

  let ext=m[1].toLowerCase();
  if(ext==="jpeg")ext="jpg";
  if(!["jpg","png","webp","gif"].includes(ext))ext="jpg";

  return{ext,base64:m[2]};
}

function slug(s){
  return String(s||"photo")
    .toLowerCase()
    .replace(/[^a-z0-9]+/g,"-")
    .replace(/^-+|-+$/g,"")
    .slice(0,55)||"photo";
}

async function existingSha(owner,repo,path,branch,token){
  const u=`https://api.github.com/repos/${encodeURIComponent(owner)}/${encodeURIComponent(repo)}/contents/${path.split("/").map(encodeURIComponent).join("/")}?ref=${encodeURIComponent(branch)}`;
  const r=await fetch(u,{headers:headers(token)});

  if(r.status===404)return null;

  let b=null;
  try{b=await r.json()}catch(_){}

  if(!r.ok)throw new Error((b&&b.message)||`GitHub HTTP ${r.status}`);
  return b.sha||null;
}

async function putFile(owner,repo,path,branch,token,content,message){
  const sha=await existingSha(owner,repo,path,branch,token);
  const payload={message,content,branch};
  if(sha)payload.sha=sha;

  return api(
    `https://api.github.com/repos/${encodeURIComponent(owner)}/${encodeURIComponent(repo)}/contents/${path.split("/").map(encodeURIComponent).join("/")}`,
    {
      method:"PUT",
      headers:headers(token),
      body:JSON.stringify(payload)
    }
  );
}

function progress(done,total,text){
  $("publishProgress").hidden=false;
  const pct=total?Math.round(done/total*100):0;
  $("progressText").textContent=text;
  $("progressPct").textContent=pct+"%";
  $("progressBar").style.width=pct+"%";
}

$("testGitHubBtn").onclick=async()=>{
  const owner=$("ghOwner").value.trim();
  const repo=$("ghRepo").value.trim();
  const branch=$("ghBranch").value.trim();
  const token=$("ghToken").value.trim();

  if(!token){
    ghStatus("Paste your fine-grained GitHub token first.","error");
    return;
  }

  try{
    const info=await api(
      `https://api.github.com/repos/${encodeURIComponent(owner)}/${encodeURIComponent(repo)}`,
      {headers:headers(token)}
    );

    await api(
      `https://api.github.com/repos/${encodeURIComponent(owner)}/${encodeURIComponent(repo)}/branches/${encodeURIComponent(branch)}`,
      {headers:headers(token)}
    );

    ghStatus(`CONNECTED\n${info.full_name}\nBranch: ${branch}\nToken can access this repository.`,"ok");
    toast("GitHub connected");
  }catch(e){
    ghStatus("CONNECTION FAILED\n"+e.message,"error");
    toast("Connection failed","error");
  }
};

$("publishGitHubBtn").onclick=async()=>{
  const owner=$("ghOwner").value.trim();
  const repo=$("ghRepo").value.trim();
  const branch=$("ghBranch").value.trim();
  const token=$("ghToken").value.trim();

  if(!token){
    ghStatus("Paste your fine-grained GitHub token first.","error");
    return;
  }

  const btn=$("publishGitHubBtn");
  btn.disabled=true;
  btn.textContent="PUBLISHING...";

  try{
    await api(
      `https://api.github.com/repos/${encodeURIComponent(owner)}/${encodeURIComponent(repo)}`,
      {headers:headers(token)}
    );

    const out=[];
    const uploads=[];
    const used=new Set();

    for(let i=0;i<items.length;i++){
      const p=items[i];
      const pt=dataParts(p.image);
      let image=p.image;

      if(pt){
        let base=slug(p.title||`photo-${i+1}`);
        let name=base+"."+pt.ext;
        let n=2;

        while(used.has(name)){
          name=base+"-"+n+"."+pt.ext;
          n++;
        }

        used.add(name);
        image="images/"+name;
        uploads.push({path:image,base64:pt.base64,title:p.title});
      }

      out.push({...p,image});
    }

    let done=0;
    const total=uploads.length+1;
    progress(done,total,"Preparing upload...");

    for(const f of uploads){
      progress(done,total,"Uploading "+f.path+"...");
      await putFile(owner,repo,f.path,branch,token,f.base64,"Gallery photo: "+f.title);
      done++;
      progress(done,total,"Uploaded "+f.path);
    }

    const gallery="window.CALLAWAY_GALLERY = "+JSON.stringify(out,null,2)+";";
    progress(done,total,"Updating gallery-data.js...");

    await putFile(
      owner,
      repo,
      "gallery-data.js",
      branch,
      token,
      b64Utf8(gallery),
      "Update Callaway JROTC gallery"
    );

    done++;
    progress(done,total,"Published successfully");

    items=out;
    render();

    ghStatus(
      `PUBLISHED SUCCESSFULLY\nPhotos uploaded: ${uploads.length}\ngallery-data.js updated.\nGitHub Pages will redeploy automatically.`,
      "ok"
    );

    toast("Gallery published");
  }catch(e){
    console.error(e);
    progress(0,1,"Publish failed");
    ghStatus("PUBLISH FAILED\n"+e.message,"error");
    toast("Publish failed","error");
  }finally{
    btn.disabled=false;
    btn.textContent="PUBLISH TO GITHUB";
  }
};

render();
